   545→// ==================== Ambient Music ====================
   546→function AmbientMusic() {
   547→  const audioRef = useRef<HTMLAudioElement>(null);
   548→  const [muted, setMuted] = useState(true);
   549→  const mountedRef = useRef(false);
   550→
   551→  useEffect(() => {
   552→    mountedRef.current = true;
   553→    const saved = localStorage.getItem('ambient-music-muted');
   554→    if (saved !== null) {
   555→      setMuted(saved === 'true');
   556→    }
   557→  }, []);
   558→
   559→  useEffect(() => {
   560→    if (!mountedRef.current || !audioRef.current) return;
   561→    audioRef.current.muted = muted;
   562→    audioRef.current.volume = 0.3;
   563→  }, [muted]);
   564→
   565→  const toggle = () => {
   566→    const next = !muted;
   567→    setMuted(next);
   568→    if (mountedRef.current) {
   569→      localStorage.setItem('ambient-music-muted', String(next));
   570→    }
   571→    if (audioRef.current && next === false) {
   572→      audioRef.current.play().catch(() => {});
   573→    }
   574→  };
   575→
   576→  return (
   577→    <>
   578→      <audio ref={audioRef} src="https://cdn.freesound.org/previews/612/612095_5674468-lq.mp3" loop preload="none" />
   579→      <Button
   580→        variant="ghost"
   581→        size="sm"
   582→        onClick={toggle}
   583→        className="text-white/40 hover:text-white/80 hover:bg-white/5 w-8 h-8 p-0"
   584→        aria-label={muted ? 'Unmute music' : 'Mute music'}
   585→      >
   586→        {muted ? <VolumeX className="w-4 h-4" /> : <Volume2 className="w-4 h-4" />}
   587→      </Button>
   588→    </>
   589→  );
   590→}
