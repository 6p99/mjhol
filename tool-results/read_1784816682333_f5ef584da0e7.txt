   107→function useGithubData() {
   108→  const [profile, setProfile] = useState<GithubProfile | null>(null);

   278→    fetchComments();
   279→    const interval = setInterval(fetchComments, 30000);
   280→    return () => clearInterval(interval);
   281→  }, [fetchComments]);
   282→
   283→  const handleSubmitComment = async () => {
   284→    if (!newComment.trim()) return;
   285→    setSending(true);
   286→    setError('');
   287→    try {
   288→      const res = await fetch('/api/comments', {
   289→        method: 'POST',
   290→        headers: { 'Content-Type': 'application/json' },
   291→        body: JSON.stringify({ content: newComment }),
   292→      });

   295→        if (res.status === 401) setError('يجب تسجيل الدخول عبر ديسكورد أولاً للتعليق');

   304→        return;
   305→      }

   307→      setError('');
   308→      setCooldownInfo(null);
