   201→    const fontSize = 14;
   202→    const cols = Math.floor(canvas.width/fontSize);
   203→    const drops = Array(cols).fill(1);
   204→    const draw = () => {
   205→      ctx.fillStyle='rgba(0,0,0,0.05)';
   206→      ctx.fillRect(0,0,canvas.width,canvas.height);
   207→      ctx.fillStyle='rgba(255,255,255,0.04)';
   208→      ctx.font=fontSize+'px monospace';
   209→      for (let i=0;i<drops.length;i++) {
   210→        const t = chars[Math.floor(Math.random()*chars.length)];
   211→        ctx.fillText(t,i*fontSize,drops[i]*fontSize);
   212→        if (drops[i]*fontSize>canvas.height&&Math.random()>0.975) drops[i]=0;
   213→        drops[i]++;
   214→      }
   215→    };
   216→    const interval = setInterval(draw,50);
   217→    return () => { clearInterval(interval); window.removeEventListener('resize',resize); };
   218→  },[]);
   219→  return <canvas ref={canvasRef} className="fixed inset-0 -z-[8] pointer-events-none hidden sm:block" />;
   220→}
