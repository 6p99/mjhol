   300→function MatrixRain() {
   301→  const canvasRef = useRef<HTMLCanvasElement>(null);
   302→  const [isVisible, setIsVisible] = useState(false);
   303→
   304→  useEffect(() => {
   305→    const checkDesktop = () => setIsVisible(window.innerWidth >= 1024);
   306→    checkDesktop();
   307→    window.addEventListener('resize', checkDesktop);
   308→    return () => window.removeEventListener('resize', checkDesktop);
   309→  }, []);
   310→
   311→  useEffect(() => {
   312→    if (!isVisible || !canvasRef.current) return;
   313→    const canvas = canvasRef.current;
   314→    const ctx = canvas.getContext('2d');
   315→    if (!ctx) return;
   316→
   317→    const resize = () => {
   318→      canvas.width = window.innerWidth;
   319→      canvas.height = window.innerHeight;
   320→    };
   321→    resize();
   322→    window.addEventListener('resize', resize);
   323→
   324→    const chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789@#$%^&*()_+-=[]{}|;:,.<>?/~`';
   325→    const fontSize = 14;
   326→    const columns = Math.floor(canvas.width / fontSize);
   327→    const drops: number[] = Array(columns).fill(1).map(() => Math.random() * -100);
   328→
   329→    const draw = () => {
   330→      ctx.fillStyle = 'rgba(0, 0, 0, 0.05)';
   331→      ctx.fillRect(0, 0, canvas.width, canvas.height);
   332→      ctx.fillStyle = 'rgba(255, 255, 255, 0.04)';
   333→      ctx.font = `${fontSize}px monospace`;
   334→
   335→      for (let i = 0; i < drops.length; i++) {
   336→        const text = chars.charAt(Math.floor(Math.random() * chars.length));
   337→        ctx.fillText(text, i * fontSize, drops[i] * fontSize);
   338→        if (drops[i] * fontSize > canvas.height && Math.random() > 0.975) {
   339→          drops[i] = 0;
   340→        }
   341→        drops[i]++;
   342→      }
   343→    };
   344→
   345→    const interval = setInterval(draw, 50);
   346→
   347→    return () => {
   348→      clearInterval(interval);
   349→      window.removeEventListener('resize', resize);
   350→    };
   351→  }, [isVisible]);
   352→
   353→  if (!isVisible) return null;
   354→
   355→  return (
   356→    <canvas
   357→      ref={canvasRef}
   358→      className="fixed inset-0 pointer-events-none z-0"
   359→      style={{ opacity: 0.04 }}
   360→    />
   361→  );
   362→}
